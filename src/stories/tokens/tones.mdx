export const SOFT_SITE = ATOM_TONES.find((site) => site.id === 'skeleton-atom-tones');
export const BORDER_SITE = FIELD_VARIANTS.find((site) => site.id === 'input-field-tones');

export const rowsOf = (site) => {
	// The tone tables exist only inside the fixed copies, so rows are read out of their class strings.
	const rows = [...(site?.fixed ?? '').matchAll(/\t\t(\w+): "([^"]+)"/g)];
	if (!rows.length) throw new Error(`[tokens/tones] no rows in ${site?.id ?? 'the tone table'}`);
	return rows.map(([, tone, row]) => ({ tone, row }));
};

export const rowOf = (site, tone) => {
	const row = site?.fixed?.match(new RegExp(`\\t\\t${tone}: "([^"]+)"`));
	if (!row) throw new Error(`[tokens/tones] no ${site?.id ?? 'tone'} row for ${tone}`);
	return row[1];
};

export const tokenFor = (tone) => {
	const token = tone === 'default' ? 'primary' : tone;
	const mapped = token in SINGLE && Boolean(SINGLE[token].source);
	const literal = Object.values(STATIC_TOKENS).some((scheme) => `${token}-foreground` in scheme);
	return mapped || literal ? token : undefined;
};

export const TONES = rowsOf(BORDER_SITE).map(({ tone, row }) => {
	const token = tokenFor(tone);
	if (!token) throw new Error(`[tokens/tones] ${tone} resolves to no token`);
	return {
		tone,
		token,
		soft: rowOf(SOFT_SITE, tone),
		// The idle border is the first class token of the row.
		border: row.split(' ')[0]
	};
});

# The seven tones

## Seven tones

A tone is one of {TONES.length} names: <code>{NAMES.join(', ')}</code>. A tone paints with the token of
the same name, except `default`, which paints with `primary` so the plain control is the brand colour.
The variant table names that brand tone `primary`; this page calls it `default` because `default` is
the prop value a consumer writes. The soft and border cells read their class strings from the
generated copy, so a change to the tone table moves this page.

<Header />

{TONES.map((row) => (

<Specimen key={row.tone} {...row} />
))}

## Text uses the derived tone

A fill and a soft surface paint with the tone itself. Text paints with the `-text` token, which mixes
the tone toward the page foreground, because the raw tone reads below AA as text. A border uses the
tone as well, except for <code>{TEXT_BORDER_TONES.join(' and ')}</code>: those edge classes carry
`-text` so the hairline clears 3:1 against the page.

{TEXT_TONES.map((tone) => (

<ToneText key={tone} tone={tone} />
))}

## Presentations above the tones

{VARIANT_COMPONENTS.length} components take a tone above their presentation: {VARIANT_COMPONENTS.join(', ')}.
The variant table carries {VARIANT_ROWS.length} rows over {PRESENTATIONS.length} presentations
({PRESENTATIONS.join(', ')}), and every one reuses the tones above.
